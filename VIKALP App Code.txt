// SAVE IMAGE TO PHONE GALLERY
// =============================================================

fun saveImageToGallery(
    imageUrl: String,
    plant: String,
    disease: String,
    timestamp: String
) {

    Thread {

        try {

            val url = URL(imageUrl)

            val connection =
                url.openConnection() as HttpURLConnection

            connection.requestMethod = "GET"

            connection.connectTimeout = 10000
            connection.readTimeout = 10000

            connection.connect()

            if (connection.responseCode != 200) {
                connection.disconnect()
                return@Thread
            }

            val bitmap =
                BitmapFactory.decodeStream(
                    connection.inputStream
                )

            connection.disconnect()

            if (bitmap == null) {
                return@Thread
            }


            // -----------------------------------------------------
            // CLEAN FOLDER / FILE NAMES
            // -----------------------------------------------------

            val cleanPlant =
                plant.replace(
                    Regex("[^a-zA-Z0-9_() -]"),
                    "_"
                )

            val cleanDisease =
                disease.replace(
                    Regex("[^a-zA-Z0-9_() -]"),
                    "_"
                )

            val cleanTimestamp =
                timestamp.replace(
                    Regex("[^a-zA-Z0-9_-]"),
                    "_"
                )


            val fileName =
                "${cleanTimestamp}_${cleanPlant}_${cleanDisease}.jpg"


            // -----------------------------------------------------
            // MEDIA STORE
            // -----------------------------------------------------

            val resolver =
                PlantDiseaseMonitorApplication
                    .instance
                    .contentResolver


            val contentValues =
                ContentValues().apply {

                    put(
                        MediaStore.Images.Media.DISPLAY_NAME,
                        fileName
                    )

                    put(
                        MediaStore.Images.Media.MIME_TYPE,
                        "image/jpeg"
                    )

                    if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {

                        put(
                            MediaStore.Images.Media.RELATIVE_PATH,
                            Environment.DIRECTORY_PICTURES +
                                    "/PlantDisease/" +
                                    cleanPlant +
                                    "/" +
                                    cleanDisease
                        )

                        put(
                            MediaStore.Images.Media.IS_PENDING,
                            1
                        )
                    }
                }


            val imageUri =
                resolver.insert(
                    MediaStore.Images.Media.EXTERNAL_CONTENT_URI,
                    contentValues
                )


            if (imageUri == null) {
                return@Thread
            }


            // -----------------------------------------------------
            // WRITE IMAGE
            // -----------------------------------------------------

            resolver.openOutputStream(imageUri).use { outputStream ->

                if (outputStream != null) {

                    bitmap.compress(
                        android.graphics.Bitmap.CompressFormat.JPEG,
                        95,
                        outputStream
                    )
                }
            }


            // -----------------------------------------------------
            // FINISH MEDIA STORE OPERATION
            // -----------------------------------------------------

            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {

                val completedValues =
                    ContentValues().apply {

                        put(
                            MediaStore.Images.Media.IS_PENDING,
                            0
                        )
                    }

                resolver.update(
                    imageUri,
                    completedValues,
                    null,
                    null
                )
            }


            // -----------------------------------------------------
            // SUCCESS MESSAGE
            // -----------------------------------------------------

            Handler(Looper.getMainLooper()).post {

                Toast.makeText(
                    PlantDiseaseMonitorApplication.instance,
                    appText(AppLanguage.ENGLISH, "saved"),
                    Toast.LENGTH_SHORT
                ).show()
            }


        } catch (e: Exception) {

            Handler(Looper.getMainLooper()).post {

                Toast.makeText(
                    PlantDiseaseMonitorApplication.instance,
                    appText(AppLanguage.ENGLISH, "saveFailed"),
                    Toast.LENGTH_SHORT
                ).show()
            }
        }

    }.start()
}

fun translatePlant(
    language: AppLanguage,
    plant: String
): String {

    if (language == AppLanguage.ENGLISH) {
        return plant
    }

    val map = mapOf(
        "Apple" to Pair("सफरचंद", "सेब"),
        "Blueberry" to Pair("ब्लूबेरी", "ब्लूबेरी"),
        "Cherry_(including_sour)" to Pair("चेरी", "चेरी"),

    fun fetchHistory() {

        Thread {

            try {

                val url =
                    URL(
                        "http://10.138.66.33:5000/history"
                    )

                val connection =
                    url.openConnection() as HttpURLConnection

                connection.requestMethod = "GET"

                connection.connectTimeout = 5000
                connection.readTimeout = 5000

                val responseCode =
                    connection.responseCode


                if (responseCode == 200) {

                    val response =
                        connection.inputStream
                            .bufferedReader()
                            .use { it.readText() }


                    val history =
                        JSONArray(response)


                    handler.post {

                        val newHistoryList =
                            mutableListOf<DetectionHistory>()


                        for (
                        i in 0 until history.length()
                        ) {

                            val item =
                                history.getJSONObject(i)


                            // -------------------------
                            // PLANT
                            // -------------------------

                            val historyPlant =
                                item.optString(
                                    "plant",
                                    "—"
                                )


                            // -------------------------
                            // DISEASE / CONDITION
                            // -------------------------

                            val historyDisease =
                                item.optString(
                                    "disease",
                                    item.optString(
                                        "condition",
                                        "—"
                                    )
                                )


                            // -------------------------
                            // CONFIDENCE
                            // -------------------------

                            var historyConfidence =
                                item.optString(
                                    "confidence",
                                    "—"
                                )

                            historyConfidence =
                                historyConfidence
                                    .replace("%", "")
                                    .trim()


                            // -------------------------
                            // STATUS
                            // -------------------------

                            val historyStatus =
                                item.optString(
                                    "status",
                                    "—"
                                )


                            // -------------------------
                            // TIMESTAMP
                            // -------------------------

                            val rawHistoryTimestamp =
                                item.optString(
                                    "timestamp",
                                    "—"
                                )

                            var historyTimestamp =
                                rawHistoryTimestamp


                            // Convert:
                            // 20260912_084304
                            //
                            // into:
                            // 2026-09-12 08:43:04

                            if (
                                historyTimestamp.matches(
                                    Regex("\\d{8}_\\d{6}")
                                )
                            ) {

                                historyTimestamp =
                                    "${historyTimestamp.substring(0, 4)}-" +
                                            "${historyTimestamp.substring(4, 6)}-" +
                                            "${historyTimestamp.substring(6, 8)} " +
                                            "${historyTimestamp.substring(9, 11)}:" +
                                            "${historyTimestamp.substring(11, 13)}:" +
                                            "${historyTimestamp.substring(13, 15)}"
                            }


                            // -------------------------
                            // HISTORICAL IMAGE
                            // -------------------------

                            var historyImageUrl =
                                item.optString(
                                    "image_url",
                                    ""
                                )

                            if (
                                historyImageUrl.isNotBlank() &&
                                historyImageUrl.startsWith("/")
                            ) {

                                historyImageUrl =
                                    "http://10.138.66.33:5000$historyImageUrl"
                            }


                            if (historyImageUrl.isBlank()) {

                                historyImageUrl =
                                    "http://10.138.66.33:5000/image?timestamp=$rawHistoryTimestamp"
                            }


                            newHistoryList.add(
                                DetectionHistory(
                                    plant = historyPlant,
                                    disease = historyDisease,
                                    confidence = historyConfidence,
                                    status = historyStatus,
                                    timestamp = historyTimestamp,
                                    imageUrl = historyImageUrl
                                )
                            )
                        }


                        historyList =
                            newHistoryList
                    }
                }


                connection.disconnect()


            } catch (e: Exception) {

                handler.post {

                    println(
                        "History error: ${e.message}"
                    )
                }
            }

        }.start()
    }

                    )

                val connection =
                    url.openConnection() as HttpURLConnection

                connection.requestMethod = "GET"

                connection.connectTimeout = 5000
                connection.readTimeout = 5000

                val responseCode =
                    connection.responseCode


                if (responseCode == 200) {

                    val response =
                        connection.inputStream
                            .bufferedReader()
                            .use { it.readText() }


                    val json =
                        JSONObject(response)


                    // =================================================
                    // PLANT
                    // =================================================

                    val newPlant =
                        json.optString(
                            "plant",
                            "Waiting for detection..."
                        )


                    // =================================================
                    // DISEASE
                    // =================================================

                    val newDisease =
                        json.optString(
                            "condition",
                            "—"
                        )


                    // =================================================
                    // CONFIDENCE
                    // =================================================

                    val newConfidence =
                        json.optString(
                            "confidence",
                            "—"
                        ).let {

                            if (it.endsWith("%")) {

                                it

                            } else {

                                "$it%"
                            }
                        }


                    // =================================================
                    // STATUS
                    // =================================================

                    val newStatus =
                        json.optString(
                            "status",
                            "Waiting"
                        )


                    // =================================================
                    // IMAGE
                    // =================================================

                    val newImageUrl =
                        "http://10.138.66.33:5000/image"


                    // =================================================
                    // TIMESTAMP
                    // =================================================

                    val rawTime =
                        json.optString(
                            "timestamp",
                            "—"
                        )


                    val newTime =
                        if (
                            rawTime.length == 15 &&
                            rawTime.contains("_")
                        ) {

                            rawTime.substring(0, 4) + "-" +
                                    rawTime.substring(4, 6) + "-" +
                                    rawTime.substring(6, 8) + " " +
                                    rawTime.substring(9, 11) + ":" +
                                    rawTime.substring(11, 13) + ":" +
                                    rawTime.substring(13, 15)

                        } else {

                            rawTime
                        }


                    // =================================================
                    // UPDATE UI
                    // =================================================

                    handler.post {

                        plant =
                            newPlant

                        disease =
                            newDisease

                        confidence =
                            newConfidence

                        status =
                            newStatus

                        detectionTime =
                            newTime


                        imageVersion =
                            System.currentTimeMillis()


                        latestImageUrl =
                            "$newImageUrl?cache=$imageVersion"


                        connected =
                            true


                        // =================================================
                        // AUTOMATICALLY SAVE NEW IMAGE
                        // =================================================

                        if (
                            rawTime.isNotBlank() &&
                            rawTime != "—" &&
                            rawTime != lastSavedTimestamp
                        ) {

                            lastSavedTimestamp =
                                rawTime


                            saveImageToGallery(
                                imageUrl =
                                    latestImageUrl,

                                plant =
                                    newPlant,

                                disease =
                                    newDisease,

                                timestamp =
                                    rawTime
                            )
                        }
                    }


                } else {

                    handler.post {

                        connected =
                            false
                    }
                }


                connection.disconnect()


            } catch (e: Exception) {

                handler.post {

                    connected =
                        false
                }
            }

        }.start()
    }

                            appText(selectedLanguage, "hideHistory")

                        else

                            appText(selectedLanguage, "viewHistory")
                )
            }


            Spacer(
                modifier =
                    Modifier.height(12.dp)
            )


            // =================================================
            // REFRESH BUTTON
            // =================================================

            Button(

                onClick = {

                    fetchLatest()

                },

                modifier =
                    Modifier.fillMaxWidth()

            ) {


                Text(

                    text =
                        appText(selectedLanguage, "refresh")
                )
            }
            // =================================================
            // LOGOUT BUTTON
            // =================================================

            Spacer(
                modifier = Modifier.height(12.dp)
            )

            Button(

                onClick = {

                    onLogout()

                },

                modifier =
                    Modifier.fillMaxWidth()

            ) {

                Text(
                    text = "LOGOUT"
                )
            }
        }
    }
